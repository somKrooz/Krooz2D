cmake_minimum_required(VERSION 3.20)
project(KroozEngine LANGUAGES C CXX)

# ==== Standards ====
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)

# ==== Options ====
option(KROOZ_BUILD_EXAMPLES "Build example projects" OFF)

# ==== Compiler Flags ====
if(MSVC)
    add_compile_definitions(
        _CRT_SECURE_NO_WARNINGS
        _GLFW_WIN32
        WIN32_LEAN_AND_MEAN
        NOMINMAX
    )

    add_compile_options(
        /MP          # Multi-processor compile
        /bigobj      # Large object support
        /wd4996      # Disable deprecated warnings
        /wd4244      # Disable int->float conversion warnings
    )
endif()

# ==== Source Files ====
file(GLOB_RECURSE KROOZ_SRC
    "Core/*.cpp"
    "Core/*.h"
    "Components/*.cpp"
    "external/GLAD/src/*.c"
    "external/GLFW/src/*.c"
)

# ==== Include Directories ====
set(KROOZ_INCLUDE
    Core/include
    Components/include
    external/GLAD/include
    external/GLFW/include
    external/CURL/include
    external/STB
    external/JSON
    external/GLFW/src
)

# ==== Library ====
add_library(krooz STATIC ${KROOZ_SRC})
target_include_directories(krooz PUBLIC ${KROOZ_INCLUDE})

target_link_libraries(krooz PUBLIC
    opengl32
    user32
    gdi32
    shell32
    kernel32
    advapi32
    crypt32
)

# ==== Output Paths (Flattened, no Debug/Release subfolders) ====
set_target_properties(krooz PROPERTIES
    OUTPUT_NAME "krooz"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    ARCHIVE_OUTPUT_DIRECTORY_DEBUG   "${CMAKE_BINARY_DIR}/lib"
    ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/lib"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    LIBRARY_OUTPUT_DIRECTORY_DEBUG   "${CMAKE_BINARY_DIR}/lib"
    LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/lib"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG   "${CMAKE_BINARY_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin"
)

# ==== Post-build Script ====
add_custom_command(TARGET krooz POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E echo "Running Krooz packaging script..."
    COMMAND ${CMAKE_COMMAND} -E env PYTHONIOENCODING=utf-8 python ${CMAKE_SOURCE_DIR}/package.py
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "Packaging Krooz2D after build"
)

# ==== Examples (Optional) ====
if(KROOZ_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
